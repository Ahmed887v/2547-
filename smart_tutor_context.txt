import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:provider/provider.dart';

import 'screens/home_screen.dart';
import 'services/learning_memory_service.dart';
import 'services/progress_service.dart';
import 'theme/app_theme.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();

  await SystemChrome.setPreferredOrientations([
    DeviceOrientation.portraitUp,
  ]);

  // Load existing lesson progress without changing its stored data.
  final progress = ProgressService();
  await progress.load();

  // Load vocabulary and correction memory separately from lesson progress.
  final learningMemory = LearningMemoryService();
  await learningMemory.load();

  runApp(
    SpeakLocalApp(
      progress: progress,
      learningMemory: learningMemory,
    ),
  );
}

class SpeakLocalApp extends StatelessWidget {
  final ProgressService progress;
  final LearningMemoryService learningMemory;

  const SpeakLocalApp({
    super.key,
    required this.progress,
    required this.learningMemory,
  });

  @override
  Widget build(BuildContext context) {
    return MultiProvider(
      providers: [
        ChangeNotifierProvider<ProgressService>.value(value: progress),
        ChangeNotifierProvider<LearningMemoryService>.value(
          value: learningMemory,
        ),
      ],
      child: MaterialApp(
        title: 'SpeakLocal',
        debugShowCheckedModeBanner: false,
        theme: buildAppTheme(),
        home: const HomeScreen(),
      ),
    );
  }
}

class PracticeScreen extends StatefulWidget {
  final String lessonId;

  const PracticeScreen({super.key, required this.lessonId});

  @override
  State<PracticeScreen> createState() => _PracticeScreenState();
}

class _PracticeScreenState extends State<PracticeScreen> {
  final _tts = TtsService();
  final _speech = SpeechService();
  final _confetti = ConfettiController(duration: const Duration(seconds: 2));

  late Lesson lesson;
  int index = 0;
  String heard = '';
  int? lastScore;
  String? lastFeedback;
  String? selectedChoice;
  bool listening = false;
  bool busy = false;
  bool finished = false;
  final List<int> scores = [];
  final Set<String> _savedPhraseKeys = {};
  final Set<String> _savedCorrectionKeys = {};
  bool speechReady = false;

  Exercise get current => lesson.exercises[index];

  @override
  void initState() {
    super.initState();
    lesson = findLesson(widget.lessonId)!;
    _bootstrap();
  }

  Future<void> _bootstrap() async {
    await _tts.init();
    speechReady = await _speech.init();
    if (mounted) setState(() {});
  }

  @override
  void dispose() {
    _tts.dispose();
    _speech.cancel();
    _confetti.dispose();
    super.dispose();
  }

  Future<void> _listenModel() async {
    setState(() => busy = true);
    await _tts.speak(current.phrase.text);
    if (mounted) setState(() => busy = false);
  }

  Future<void> _toggleMic() async {
    if (listening) {
      await _speech.stop();
      if (!mounted) return;
      setState(() => listening = false);
      await _evaluate();
      return;
    }

    if (!speechReady) {
      speechReady = await _speech.init();
      if (!mounted) return;
      if (!speechReady) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text(
              'Speech recognition unavailable. Enable Google voice typing / English speech pack.',
            ),
          ),
        );
        return;
      }
    }

    if (current.type == ExerciseType.chooseAndSay && selectedChoice == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Pick the correct phrase first.')),
      );
      return;
    }

    setState(() {
      listening = true;
      heard = '';
      lastScore = null;
      lastFeedback = null;
    });

    await _speech.start(
      onResult: (words, conf) {
        if (!mounted) return;
        setState(() => heard = words);
      },
    );
  }

  // Evaluate the answer and save useful learning items for later review.
  Future<void> _evaluate() async {
    final phrase = current.phrase;
    final target = current.type == ExerciseType.chooseAndSay
        ? (selectedChoice ?? phrase.text)
        : phrase.text;

    final memory = context.read<LearningMemoryService>();

    // Save each lesson phrase once per screen session.
    final phraseKey = '${lesson.id}:$index';
    if (_savedPhraseKeys.add(phraseKey)) {
      await memory.recordVocabulary(
        word: phrase.text,
        translation: phrase.meaningAr,
        example: phrase.tip.isEmpty
            ? phrase.text
            : '${phrase.text}\nTip: ${phrase.tip}',
      );
    }

    // Save only corrections confirmed by the lesson's answer key.
    if (current.type == ExerciseType.chooseAndSay &&
        selectedChoice != phrase.text) {
      if (!mounted) return;

      setState(() {
        lastScore = 20;
        lastFeedback = 'Wrong choice — pick the natural phrase.';
      });
      scores.add(20);

      await memory.recordCorrection(
        original: selectedChoice ?? '',
        corrected: phrase.text,
        explanation: 'This lesson marked the selected phrase as incorrect. '
            'Choose the phrase that matches the lesson.',
        example: '${phrase.meaningAr} — ${phrase.text}',
      );

      if (mounted) {
        await context.read<ProgressService>().loseHeart();
      }
      return;
    }

    final score = PronunciationScorer.score(target, heard);
    if (!mounted) return;

    setState(() {
      lastScore = score;
      lastFeedback = PronunciationScorer.feedback(score);
    });
    scores.add(score);

// Persist this evaluated exercise attempt for progress analytics.
// This score reflects the app's existing evaluation, not acoustic analysis.
    await PracticeAnalyticsService().recordAttempt(
      lessonId: lesson.id,
      prompt: target,
      target: target,
      score: score,
      exerciseType: 'practice',
    );

    // Save low-scoring speech transcripts for targeted future review.
    // This compares recognized words; it is not acoustic pronunciation analysis.
    if (score < 75 && heard.trim().isNotEmpty) {
      final correctionKey = '${lesson.id}:$index:${heard.trim().toLowerCase()}';

      if (_savedCorrectionKeys.add(correctionKey)) {
        await memory.recordCorrection(
          original: heard.trim(),
          corrected: target,
          explanation:
              'The recognized words did not closely match the target phrase.',
          example: '${phrase.meaningAr} — $target',
        );
      }
    }

    // The screen may have been disposed while an awaited save was running.
    if (!mounted) return;

    if (score < 45) {
      await context.read<ProgressService>().loseHeart();
    }
  }

  Future<void> _next() async {
    if (lastScore == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Speak the phrase before continuing.')),
      );
      return;
    }

    if (index < lesson.exercises.length - 1) {
      setState(() {
        index += 1;
        heard = '';
        lastScore = null;
        lastFeedback = null;
        selectedChoice = null;
        listening = false;
      });
      return;
    }

    final avg = scores.isEmpty
        ? 0
        : (scores.reduce((a, b) => a + b) / scores.length).round();
    final xp =
        ((lesson.xpReward * (avg / 100)).round()).clamp(5, lesson.xpReward);
    await context.read<ProgressService>().addPracticeResult(
          lessonId: lesson.id,
          averageScore: avg,
          earnedXp: xp,
        );
    _confetti.play();
    setState(() => finished = true);
  }

  @override
  Widget build(BuildContext context) {
    final color = Color(lesson.color.argb);
    final step = (index + 1) / lesson.exercises.length;

    if (finished) {

    }

    return Scaffold(

              ),
            ),

        ],
      ),
      body: SafeArea(

                  ),
                ),
              ),
              const SizedBox(height: 18),
              Expanded(
                child: SingleChildScrollView(
                  child: Column(
                    children: [
                      Container(

                        padding: const EdgeInsets.all(22),
                        decoration: BoxDecoration(
                          color: color.withOpacity(0.08),
                          borderRadius: BorderRadius.circular(28),
                          border: Border.all(
                              color: color.withOpacity(0.3), width: 2),
                        ),
                        child: Column(
                          children: [
                            Text(
                              _typeLabel(current.type),
                              style: TextStyle(

                              ),
                            ),
                            const SizedBox(height: 14),
                            Text(

                              textAlign: TextAlign.center,
                              style: const TextStyle(

// SpeakLocal: local vocabulary and correction review.
// All review data is stored by LearningMemoryService on this device.

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../services/learning_memory_service.dart';
import '../services/practice_analytics_service.dart';
import '../theme/app_theme.dart';
import '../widgets/stat_chip.dart';

class ReviewScreen extends StatefulWidget {
  const ReviewScreen({super.key});

  @override
  State<ReviewScreen> createState() => _ReviewScreenState();
}

class _ReviewScreenState extends State<ReviewScreen> {
  // Load saved practice analytics once when the review screen is created.
  late final PracticeAnalyticsService _analytics;
  late final Future<void> _analyticsFuture;

  @override
  void initState() {
    super.initState();
    _analytics = PracticeAnalyticsService();
    _analytics.addListener(_onAnalyticsChanged);
    _analyticsFuture = _analytics.load();
  }

  void _onAnalyticsChanged() {
    if (mounted) setState(() {});
  }

  @override
  void dispose() {
    _analytics.removeListener(_onAnalyticsChanged);
    _analytics.dispose();
    super.dispose();
  }

  // 0 = all items, 1 = vocabulary, 2 = corrections.
  int _filter = 0;
  int _index = 0;
  bool _revealed = false;

  // Hide successfully reviewed cards for this screen session only.
  // The vocabulary remains saved and can return when it is due again.
  final Set<String> _reviewedThisSession = <String>{};

  // Rank weak practice phrases first without changing saved review dates.
  List<LearningMemoryItem> _filteredItems(
    LearningMemoryService memory,
  ) {
    final weakPrompts = _analytics.summary.weakPrompts
        .map((prompt) => prompt.trim().toLowerCase())
        .where((prompt) => prompt.isNotEmpty)
        .toSet();

    bool isWeak(LearningMemoryItem item) {
      final original = item.original.trim().toLowerCase();
      final corrected = item.corrected.trim().toLowerCase();
      return weakPrompts.contains(original) || weakPrompts.contains(corrected);
    }

    final now = DateTime.now();
    final items = memory.items
        .where((item) => !_reviewedThisSession.contains(item.id))
        .where((item) {
      if (_filter == 1) {
        return item.kind == LearningMemoryKind.vocabulary;
      }
      if (_filter == 2) {
        return item.kind == LearningMemoryKind.correction;
      }
      return true;
    }).toList();

    // Sort by: overdue first, weak practice matches next, then oldest due date.
    items.sort((a, b) {
      final aDue = !a.nextReviewAt.isAfter(now);
      final bDue = !b.nextReviewAt.isAfter(now);

      if (aDue != bDue) return aDue ? -1 : 1;

      final aWeak = isWeak(a);
      final bWeak = isWeak(b);
      if (aWeak != bWeak) return aWeak ? -1 : 1;

      if (aDue && bDue) {
        return a.nextReviewAt.compareTo(b.nextReviewAt);
      }

      return b.createdAt.compareTo(a.createdAt);
    });

    return items;
  }

  Future<void> _review(
    LearningMemoryService memory,
    LearningMemoryItem item,
    int quality,
  ) async {
    // A due item moves behind the remaining due items after review.
    // Keep the index in place so the next due item is not skipped.
    final wasDue = !item.nextReviewAt.isAfter(DateTime.now());

    await memory.markReviewed(itemId: item.id, quality: quality);

    if (!mounted) return;

    setState(() {
      if (quality >= 3) {
        _reviewedThisSession.add(item.id);
      } else if (!wasDue) {
        _index++;
      }
      _revealed = false;
    });

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          quality < 3
              ? 'Saved. This item will return for review soon.'
              : 'Review saved. Keep up the good work!',
        ),
        duration: const Duration(seconds: 2),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final memory = context.watch<LearningMemoryService>();
    final items = _filteredItems(memory);
    final dueCount = memory.dueItems.length;

    // Keep the selected index valid when items or filters change.
    final safeIndex = items.isEmpty ? 0 : _index % items.length;
    final item = items.isEmpty ? null : items[safeIndex];

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'Learning review',
          style: TextStyle(fontWeight: FontWeight.w900),
        ),
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(20, 12, 20, 28),
        children: [
          Container(
            padding: const EdgeInsets.all(20),
            decoration: BoxDecoration(
              gradient: const LinearGradient(
                colors: [Color(0xFFE7F8EE), Color(0xFFD3F0E0)],
              ),
              borderRadius: BorderRadius.circular(24),
              border: Border.all(color: AppColors.green.withOpacity(0.35)),
            ),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text(
                  'Little steps, lasting progress',
                  style: TextStyle(
                    fontSize: 21,
                    fontWeight: FontWeight.w900,
                  ),
                ),
                const SizedBox(height: 8),
                const Text(
                  'Practise saved words and corrections. '
                  'Your review history stays on this device.',
                  style: TextStyle(height: 1.4, color: AppColors.ink),
                ),

import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';

/// A persisted record of one learning attempt.
///
/// The score measures the app's existing exercise evaluation. It must not be
/// interpreted as a clinical or acoustic measurement of pronunciation.
@immutable
class PracticeAttempt {
  const PracticeAttempt({
    required this.lessonId,
    required this.prompt,
    required this.target,
    required this.score,
    required this.createdAt,
    required this.exerciseType,
  });

  final String lessonId;
  final String prompt;
  final String target;
  final int score;
  final DateTime createdAt;
  final String exerciseType;

  Map<String, Object?> toJson() => {
        'lessonId': lessonId,
        'prompt': prompt,
        'target': target,
        'score': score,
        'createdAt': createdAt.toUtc().toIso8601String(),
        'exerciseType': exerciseType,
      };

  factory PracticeAttempt.fromJson(Map<String, dynamic> json) {
    final rawScore = json['score'];
    final parsedScore = rawScore is num ? rawScore.toInt() : 0;

    return PracticeAttempt(
      lessonId: json['lessonId'] as String? ?? '',
      prompt: json['prompt'] as String? ?? '',
      target: json['target'] as String? ?? '',
      score: parsedScore.clamp(0, 100),
      createdAt: DateTime.tryParse(json['createdAt'] as String? ?? '') ??
          DateTime.fromMillisecondsSinceEpoch(0, isUtc: true),
      exerciseType: json['exerciseType'] as String? ?? 'unknown',
    );
  }
}

/// Summary calculated from saved attempts rather than guessed from XP.
@immutable
class PracticeAnalyticsSummary {
  const PracticeAnalyticsSummary({
    required this.totalAttempts,
    required this.averageScore,
    required this.weakPrompts,
    required this.recentAttempts,
  });

  final int totalAttempts;
  final double averageScore;
  final List<String> weakPrompts;
  final List<PracticeAttempt> recentAttempts;
}

/// Stores exercise history separately from the existing progress and memory
/// keys, so adding analytics never overwrites the user's current progress.
class PracticeAnalyticsService extends ChangeNotifier {
  PracticeAnalyticsService({SharedPreferences? preferences})
      : _preferences = preferences;

  static const storageKey = 'speaklocal_practice_attempts_v1';
  static const int maxStoredAttempts = 2000;

  SharedPreferences? _preferences;
  final List<PracticeAttempt> _attempts = [];
  bool _loaded = false;

  List<PracticeAttempt> get attempts => List.unmodifiable(_attempts);
  bool get isLoaded => _loaded;

  Future<void> load() async {
    _preferences ??= await SharedPreferences.getInstance();
    final raw = _preferences!.getString(storageKey);
    _attempts.clear();

    if (raw != null && raw.isNotEmpty) {
      try {
        final decoded = jsonDecode(raw);
        if (decoded is List) {
          for (final entry in decoded) {
            if (entry is Map<String, dynamic>) {
              _attempts.add(PracticeAttempt.fromJson(entry));
            } else if (entry is Map) {
              _attempts.add(
                PracticeAttempt.fromJson(Map<String, dynamic>.from(entry)),
              );
            }
          }
        }
      } on FormatException {
        // Preserve app availability if this new analytics key is malformed.
        // Do not clear or modify any other saved user data.
        _attempts.clear();
      }
    }

    _attempts.sort((a, b) => b.createdAt.compareTo(a.createdAt));
    _loaded = true;
    notifyListeners();
  }

  Future<void> recordAttempt({
    required String lessonId,
    required String prompt,
    required String target,
    required int score,
    required String exerciseType,
    DateTime? createdAt,
  }) async {
    if (!_loaded) await load();

    final normalizedScore = score.clamp(0, 100);
    _attempts.insert(
      0,
      PracticeAttempt(
        lessonId: lessonId,
        prompt: prompt.trim(),
        target: target.trim(),
        score: normalizedScore,
        createdAt: (createdAt ?? DateTime.now()).toUtc(),
        exerciseType: exerciseType,
      ),
    );

    if (_attempts.length > maxStoredAttempts) {
      _attempts.removeRange(maxStoredAttempts, _attempts.length);
    }

    await _persist();
    notifyListeners();
  }

  PracticeAnalyticsSummary get summary {
    if (_attempts.isEmpty) {
      return const PracticeAnalyticsSummary(
        totalAttempts: 0,
        averageScore: 0,
        weakPrompts: [],
        recentAttempts: [],
      );
    }

    final total = _attempts.fold<int>(0, (sum, item) => sum + item.score);
    final byPrompt = <String, List<int>>{};

    for (final attempt in _attempts) {
      final key = '${attempt.lessonId}::${attempt.prompt.toLowerCase()}';
      byPrompt.putIfAbsent(key, () => []).add(attempt.score);
    }

    final weak = byPrompt.entries
        .map((entry) {
          final average =
              entry.value.reduce((a, b) => a + b) / entry.value.length;
          return (prompt: entry.key.split('::').last, average: average);
        })
        .where((item) => item.average < 70)
        .toList()
      ..sort((a, b) => a.average.compareTo(b.average));

    return PracticeAnalyticsSummary(
      totalAttempts: _attempts.length,
      averageScore: total / _attempts.length,
      weakPrompts: List.unmodifiable(weak.map((item) => item.prompt).toSet()),
      recentAttempts: List.unmodifiable(_attempts.take(10)),
    );
  }

  Future<void> _persist() async {
    final preferences = _preferences ??= await SharedPreferences.getInstance();
    await preferences.setString(
      storageKey,
      jsonEncode(_attempts.map((attempt) => attempt.toJson()).toList()),
    );
  }
}

// Local learning memory for SpeakLocal.
//
// Stores vocabulary and language corrections independently from lesson progress.
// Data stays on this device and survives app restarts.
// A future AI conversation service can call these methods to save learning items.

import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';

enum LearningMemoryKind {
  vocabulary,
  correction,
}

class LearningMemoryItem {
  final String id;
  final LearningMemoryKind kind;
  final String original;
  final String corrected;
  final String explanation;
  final String example;
  final int occurrences;
  final int successfulUses;
  final int reviewIntervalDays;
  final int correctReviewStreak;
  final DateTime createdAt;
  final DateTime? lastReviewedAt;
  final DateTime nextReviewAt;

  const LearningMemoryItem({
    required this.id,
    required this.kind,
    required this.original,
    required this.corrected,
    required this.explanation,
    required this.example,
    required this.occurrences,
    required this.successfulUses,
    required this.reviewIntervalDays,
    required this.correctReviewStreak,
    required this.createdAt,
    required this.lastReviewedAt,
    required this.nextReviewAt,
  });

  factory LearningMemoryItem.fromJson(Map<String, dynamic> json) {
    return LearningMemoryItem(
      id: json['id'] as String,
      kind: LearningMemoryKind.values.firstWhere(
        (value) => value.name == json['kind'],
        orElse: () => LearningMemoryKind.vocabulary,
      ),
      original: json['original'] as String? ?? '',
      corrected: json['corrected'] as String? ?? '',
      explanation: json['explanation'] as String? ?? '',
      example: json['example'] as String? ?? '',
      occurrences: (json['occurrences'] as num?)?.toInt() ?? 1,
      successfulUses: (json['successfulUses'] as num?)?.toInt() ?? 0,
      reviewIntervalDays:
          (json['reviewIntervalDays'] as num?)?.toInt() ?? 0,
      correctReviewStreak:
          (json['correctReviewStreak'] as num?)?.toInt() ?? 0,
      createdAt: DateTime.tryParse(json['createdAt'] as String? ?? '') ??
          DateTime.now(),
      lastReviewedAt:
          DateTime.tryParse(json['lastReviewedAt'] as String? ?? ''),
      nextReviewAt:
          DateTime.tryParse(json['nextReviewAt'] as String? ?? '') ??
              DateTime.now(),
    );
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'kind': kind.name,
        'original': original,
        'corrected': corrected,
        'explanation': explanation,
        'example': example,
        'occurrences': occurrences,
        'successfulUses': successfulUses,
        'reviewIntervalDays': reviewIntervalDays,
        'correctReviewStreak': correctReviewStreak,
        'createdAt': createdAt.toIso8601String(),
        'lastReviewedAt': lastReviewedAt?.toIso8601String(),
        'nextReviewAt': nextReviewAt.toIso8601String(),
      };

  LearningMemoryItem copyWith({
    String? original,
    String? corrected,
    String? explanation,
    String? example,
    int? occurrences,
    int? successfulUses,
    int? reviewIntervalDays,
    int? correctReviewStreak,
    DateTime? lastReviewedAt,
    DateTime? nextReviewAt,
  }) {
    return LearningMemoryItem(
      id: id,
      kind: kind,
      original: original ?? this.original,
      corrected: corrected ?? this.corrected,
      explanation: explanation ?? this.explanation,
      example: example ?? this.example,
      occurrences: occurrences ?? this.occurrences,
      successfulUses: successfulUses ?? this.successfulUses,
      reviewIntervalDays:
          reviewIntervalDays ?? this.reviewIntervalDays,
      correctReviewStreak:
          correctReviewStreak ?? this.correctReviewStreak,
      createdAt: createdAt,
      lastReviewedAt: lastReviewedAt ?? this.lastReviewedAt,
      nextReviewAt: nextReviewAt ?? this.nextReviewAt,
    );
  }
}

class LearningMemoryService extends ChangeNotifier {
  static const String _storageKey = 'speaklocal_learning_memory_v1';

  final Map<String, LearningMemoryItem> _items = {};
  bool _loaded = false;

  bool get isLoaded => _loaded;

  List<LearningMemoryItem> get items {
    final result = _items.values.toList();
    result.sort((a, b) => b.createdAt.compareTo(a.createdAt));
    return List.unmodifiable(result);
  }

  List<LearningMemoryItem> get vocabulary => items
      .where((item) => item.kind == LearningMemoryKind.vocabulary)
      .toList();

  List<LearningMemoryItem> get corrections => items
      .where((item) => item.kind == LearningMemoryKind.correction)
      .toList();

  List<LearningMemoryItem> get dueItems {
    final now = DateTime.now();
    final result = _items.values
        .where((item) => !item.nextReviewAt.isAfter(now))
        .toList();
    result.sort((a, b) => a.nextReviewAt.compareTo(b.nextReviewAt));
    return result;
  }

  Future<void> load() async {
    if (_loaded) return;

    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_storageKey);

    if (raw != null) {
      try {
        final decoded = jsonDecode(raw);
        if (decoded is List) {
          for (final entry in decoded) {
            if (entry is Map<String, dynamic>) {
              final item = LearningMemoryItem.fromJson(entry);
              _items[item.id] = item;
            } else if (entry is Map) {
              final item = LearningMemoryItem.fromJson(
                Map<String, dynamic>.from(entry),
              );
              _items[item.id] = item;
            }
          }
        }
      } on FormatException {
        // Keep the app usable if the stored learning-memory JSON is damaged.
        // Do not delete or overwrite the raw value automatically.
      }
    }

    _loaded = true;
    notifyListeners();
  }

  Future<void> recordVocabulary({
    required String word,
    String translation = '',
    String example = '',
  }) async {
    final value = word.trim();
    if (value.isEmpty) return;

    final existing = _find(
      kind: LearningMemoryKind.vocabulary,
      original: value,
    );

    if (existing != null) {
      _items[existing.id] = existing.copyWith(
        occurrences: existing.occurrences + 1,
        explanation:
            translation.trim().isEmpty ? null : translation.trim(),
        example: example.trim().isEmpty ? null : example.trim(),
      );
    } else {
      final now = DateTime.now();
      final item = LearningMemoryItem(
        id: 'v_${now.microsecondsSinceEpoch}',
        kind: LearningMemoryKind.vocabulary,
        original: value,
        corrected: value,
        explanation: translation.trim(),
        example: example.trim(),
        occurrences: 1,
        successfulUses: 0,
        reviewIntervalDays: 0,
        correctReviewStreak: 0,
        createdAt: now,
        lastReviewedAt: null,
        nextReviewAt: now,
      );
      _items[item.id] = item;
    }

    await _persist();
  }

  Future<void> recordCorrection({
    required String original,
    required String corrected,
    String explanation = '',
    String example = '',
  }) async {
    final wrong = original.trim();
    final right = corrected.trim();
    if (wrong.isEmpty || right.isEmpty) return;

    final existing = _find(
      kind: LearningMemoryKind.correction,
      original: wrong,
      corrected: right,
    );

    if (existing != null) {
      _items[existing.id] = existing.copyWith(
        occurrences: existing.occurrences + 1,
        explanation:
            explanation.trim().isEmpty ? null : explanation.trim(),
        example: example.trim().isEmpty ? null : example.trim(),
      );
    } else {
      final now = DateTime.now();
      final item = LearningMemoryItem(
        id: 'c_${now.microsecondsSinceEpoch}',
        kind: LearningMemoryKind.correction,
        original: wrong,
        corrected: right,
        explanation: explanation.trim(),
        example: example.trim(),
        occurrences: 1,
        successfulUses: 0,
        reviewIntervalDays: 0,
        correctReviewStreak: 0,
        createdAt: now,
        lastReviewedAt: null,
        nextReviewAt: now,
      );
      _items[item.id] = item;
    }

    await _persist();
  }

  // Call this when the learner independently uses a saved correction correctly.
  Future<void> recordSuccessfulUse(String itemId) async {
    final item = _items[itemId];
    if (item == null) return;

    _items[itemId] = item.copyWith(
      successfulUses: item.successfulUses + 1,
    );
    await _persist();
  }

  // Quality scale: 0 = forgot completely, 5 = remembered easily.
  // Repeated successful reviews gradually increase the interval.
  Future<void> markReviewed({
    required String itemId,
    required int quality,
  }) async {
    final item = _items[itemId];
    if (item == null) return;

    final score = quality.clamp(0, 5);
    final int interval;
    final int streak;

    if (score < 3) {
      interval = 1;
      streak = 0;
    } else {
      interval = item.reviewIntervalDays == 0
          ? 1
          : score == 3
              ? item.reviewIntervalDays
              : item.reviewIntervalDays * 2;
      streak = item.correctReviewStreak + 1;
    }

    final now = DateTime.now();
    _items[itemId] = item.copyWith(
      reviewIntervalDays: interval,
      correctReviewStreak: streak,
      lastReviewedAt: now,
      nextReviewAt: now.add(Duration(days: interval)),
    );

    await _persist();
  }

  LearningMemoryItem? _find({
    required LearningMemoryKind kind,
    required String original,
    String? corrected,
  }) {
    final normalizedOriginal = _normalize(original);
    final normalizedCorrected =
        corrected == null ? null : _normalize(corrected);

    for (final item in _items.values) {
      if (item.kind != kind ||
          _normalize(item.original) != normalizedOriginal) {
        continue;
      }

      if (normalizedCorrected != null &&
          _normalize(item.corrected) != normalizedCorrected) {
        continue;
      }

      return item;
    }
    return null;
  }

  String _normalize(String value) =>
      value.toLowerCase().replaceAll(RegExp(r'\s+'), ' ').trim();

  Future<void> _persist() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
      _storageKey,
      jsonEncode(_items.values.map((item) => item.toJson()).toList()),
    );
    notifyListeners();
  }
}

import 'dart:async';

import 'package:speech_to_text/speech_to_text.dart';

/// Handles speech recognition while preserving the transcript across sessions.
/// Bluetooth input depends on Android's active route and recognizer support.
class SpeechService {
  final SpeechToText _speech = SpeechToText();

  bool available = false;
  String lastWords = '';
  double confidence = 0;

  bool _keepListening = false;
  bool _starting = false;
  bool _disposed = false;
  bool _restartPending = false;

  Timer? _restartTimer;
  String _partialWords = '';
  final List<String> _completedSegments = [];
  void Function(String words, double confidence)? _onResult;

  Future<bool> init() async {
    if (available) return true;

    available = await _speech.initialize(
      onError: (_) {
        // Status events may follow an error. Restart only while recording
        // remains enabled by the user.
      },
      onStatus: _onStatus,
    );

    return available;
  }

  bool get isListening => _keepListening || _starting || _speech.isListening;

  Future<void> start({
    required void Function(String words, double confidence) onResult,
  }) async {
    if (!available) {
      await init();
    }
    if (!available) return;

    lastWords = '';
    confidence = 0;

    await _speech.listen(
      onResult: (result) {
        lastWords = result.recognizedWords;
        confidence = result.confidence > 0 ? result.confidence : confidence;
        onResult(lastWords, confidence);
      },
      listenOptions: SpeechListenOptions(
        listenMode: ListenMode.dictation,
        cancelOnError: false,
        partialResults: true,
        localeId: 'en_US',
        listenFor: const Duration(minutes: 5),
        pauseFor: const Duration(seconds: 25),
      ),
    );
  }

  Future<void> _startSession() async {
    if (_disposed || !_keepListening || _starting) return;
    if (_speech.isListening) return;

    _starting = true;

    try {
      await _speech.listen(
        onResult: (result) {
          if (_disposed || !_keepListening) return;

          final words = result.recognizedWords.trim();

          if (result.finalResult) {
            // Avoid appending the same final segment more than once.
            if (words.isNotEmpty &&
                (_completedSegments.isEmpty ||
                    _completedSegments.last != words)) {
              _completedSegments.add(words);
            }
            _partialWords = '';
          } else {
            _partialWords = words;
          }

          lastWords = [
            ..._completedSegments,
            if (_partialWords.isNotEmpty) _partialWords,
          ].join(' ').replaceAll(RegExp(r'\s+'), ' ').trim();

          if (result.confidence > 0) {
            confidence = result.confidence;
          }

          _onResult?.call(lastWords, confidence);
        },
        listenOptions: SpeechListenOptions(
          listenMode: ListenMode.dictation,
          cancelOnError: false,
          partialResults: true,
          localeId: 'en_US',
          listenFor: const Duration(minutes: 5),
          pauseFor: const Duration(seconds: 25),
        ),
      );
    } catch (_) {
      _scheduleRestart();
    } finally {
      _starting = false;
    }
  }

  void _onStatus(String status) {
    if (_disposed || !_keepListening) return;

    final normalized = status.toLowerCase();
    if (normalized == 'done' || normalized == 'notlistening') {
      _scheduleRestart();
    }
  }

  void _scheduleRestart() {
    if (_disposed || !_keepListening || _restartPending) return;

    _restartPending = true;
    _restartTimer?.cancel();

    _restartTimer = Timer(const Duration(milliseconds: 700), () async {
      _restartPending = false;

      if (_disposed || !_keepListening) return;

      await _startSession();
    });
  }

  Future<void> stop() async {
    // Turn off restart first, because Android may emit "done" after stop().
    _keepListening = false;
    _restartTimer?.cancel();
    _restartPending = false;

    if (_speech.isListening) {
      await _speech.stop();
    }

    lastWords = [
      ..._completedSegments,
      if (_partialWords.isNotEmpty) _partialWords,
    ].join(' ').replaceAll(RegExp(r'\s+'), ' ').trim();

    _onResult?.call(lastWords, confidence);
  }

  Future<void> cancel() async {
    _keepListening = false;
    _restartTimer?.cancel();
    _restartPending = false;

    _completedSegments.clear();
    _partialWords = '';
    lastWords = '';
    _onResult = null;

    if (_speech.isListening) {
      await _speech.cancel();
    }
  }

  void dispose() {
    _disposed = true;
    _keepListening = false;
    _restartTimer?.cancel();
    _speech.cancel();
  }
}

/// Estimates transcript similarity; it does not measure pronunciation acoustics.
class PronunciationScorer {
  static int score(String target, String spoken) {
    final a = _normalize(target);
    final b = _normalize(spoken);

    if (a.isEmpty || b.isEmpty) return 0;
    if (a == b) return 100;

    final aw = a.split(' ').where((w) => w.isNotEmpty).toList();
    final bw = b.split(' ').where((w) => w.isNotEmpty).toList();

    if (aw.isEmpty || bw.isEmpty) return 0;

    final wordDistance = _wordLevenshtein(aw, bw);
    final maxWords = aw.length > bw.length ? aw.length : bw.length;
    final wordSimilarity = 1 - wordDistance / maxWords;
    final characterSimilarity = _similarity(a, b);

    return ((wordSimilarity * 0.75 + characterSimilarity * 0.25) * 100)
        .round()
        .clamp(0, 100);
  }

  static String feedback(int score) {
    if (score >= 90) {
      return 'Excellent! The recognized words closely match the target.';
    }
    if (score >= 75) {
      return 'Great job! The recognized words mostly match.';
    }
    if (score >= 55) {
      return 'Good try — listen again and speak a little more slowly.';
    }
    if (score >= 35) {
      return 'Keep going — listen to the model, then try again.';
    }
    return 'Try once more. Tap Listen first, then speak clearly.';
  }

  static String _normalize(String value) {
    return value
        .toLowerCase()
        .replaceAll('’', "'")
        .replaceAll(RegExp(r'\bcan not\b'), 'cannot')
        .replaceAll(RegExp(r'\bgonna\b'), 'going to')
        .replaceAll(RegExp(r'\bwanna\b'), 'want to')
        .replaceAll(RegExp(r"[^a-z0-9\s']"), ' ')
        .replaceAll(RegExp(r'\s+'), ' ')
        .trim();
  }

  static double _similarity(String a, String b) {
    if (a == b) return 1;
    if (a.isEmpty || b.isEmpty) return 0;

    final distance = _levenshtein(a, b);
    final maxLength = a.length > b.length ? a.length : b.length;

    return 1 - distance / maxLength;
  }

  static int _wordLevenshtein(List<String> a, List<String> b) {
    final dp = List.generate(
      a.length + 1,
      (_) => List<int>.filled(b.length + 1, 0),
    );

    for (var i = 0; i <= a.length; i++) {
      dp[i][0] = i;
    }
    for (var j = 0; j <= b.length; j++) {
      dp[0][j] = j;
    }

    for (var i = 1; i <= a.length; i++) {
      for (var j = 1; j <= b.length; j++) {
        final cost = a[i - 1] == b[j - 1] ? 0 : 1;

        dp[i][j] = [
          dp[i - 1][j] + 1,
          dp[i][j - 1] + 1,
          dp[i - 1][j - 1] + cost,
        ].reduce((x, y) => x < y ? x : y);
      }
    }

    return dp[a.length][b.length];
  }

  static int _levenshtein(String a, String b) {
    final dp = List.generate(
      a.length + 1,
      (_) => List<int>.filled(b.length + 1, 0),
    );

    for (var i = 0; i <= a.length; i++) {
      dp[i][0] = i;
    }
    for (var j = 0; j <= b.length; j++) {
      dp[0][j] = j;
    }

    for (var i = 1; i <= a.length; i++) {
      for (var j = 1; j <= b.length; j++) {
        final cost = a[i - 1] == b[j - 1] ? 0 : 1;

        dp[i][j] = [
          dp[i - 1][j] + 1,
          dp[i][j - 1] + 1,
          dp[i - 1][j - 1] + cost,
        ].reduce((x, y) => x < y ? x : y);
      }
    }

    return dp[a.length][b.length];
  }
}

class TtsService {

  Future<void> init() async {

  }

  Future<void> stop() => _tts.stop();

  void dispose() {
    _tts.stop();
  }
}

import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';

import '../data/lessons.dart';

class ProgressService extends ChangeNotifier {
  static const _key = 'speaklocal_progress_v1';

  int xp = 0;
  int streak = 0;
  int hearts = 5;
  int dailyGoal = 50;
  int dailyXp = 0;
  String? lastPracticeDate;
  Set<String> completedLessons = {};
  Map<String, int> bestScores = {};

  bool get dailyGoalMet => dailyXp >= dailyGoal;

  Future<void> load() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_key);
    if (raw == null) {
      notifyListeners();
      return;
    }
    final map = jsonDecode(raw) as Map<String, dynamic>;
    xp = map['xp'] as int? ?? 0;
    streak = map['streak'] as int? ?? 0;
    hearts = map['hearts'] as int? ?? 5;
    dailyGoal = map['dailyGoal'] as int? ?? 50;
    dailyXp = map['dailyXp'] as int? ?? 0;
    lastPracticeDate = map['lastPracticeDate'] as String?;
    completedLessons = Set<String>.from(
      (map['completedLessons'] as List?)?.cast<String>() ?? [],
    );
    final rawScores = map['bestScores'];
    if (rawScores is Map) {
      bestScores = {
        for (final e in rawScores.entries)
          e.key.toString(): (e.value as num).toInt(),
      };
    } else {
      bestScores = {};
    }
    _rollDailyIfNeeded();
    notifyListeners();
  }

  Future<void> _save() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
      _key,
      jsonEncode({
        'xp': xp,
        'streak': streak,
        'hearts': hearts,
        'dailyGoal': dailyGoal,
        'dailyXp': dailyXp,
        'lastPracticeDate': lastPracticeDate,
        'completedLessons': completedLessons.toList(),
        'bestScores': bestScores,
      }),
    );
  }

  String _today() {
    final n = DateTime.now();
    return '${n.year}-${n.month.toString().padLeft(2, '0')}-${n.day.toString().padLeft(2, '0')}';
  }

  void _rollDailyIfNeeded() {
    final today = _today();
    if (lastPracticeDate == null) return;
    if (lastPracticeDate == today) return;

    final last = DateTime.tryParse(lastPracticeDate!);
    final now = DateTime.now();
    if (last != null) {
      final gap = DateTime(now.year, now.month, now.day)
          .difference(DateTime(last.year, last.month, last.day))
          .inDays;
      if (gap > 1) streak = 0;
    }
    dailyXp = 0;
  }

  bool isLessonUnlocked(String lessonId) {
    final lessons = allLessons();
    final index = lessons.indexWhere((l) => l.id == lessonId);
    if (index <= 0) return true;
    return completedLessons.contains(lessons[index - 1].id);
  }

  bool isCompleted(String lessonId) => completedLessons.contains(lessonId);

  Future<void> addPracticeResult({
    required String lessonId,
    required int averageScore,
    required int earnedXp,
  }) async {
    _rollDailyIfNeeded();
    final today = _today();
    if (lastPracticeDate != today) {
      if (lastPracticeDate != null) {
        final last = DateTime.tryParse(lastPracticeDate!);
        final now = DateTime.now();
        if (last != null) {
          final gap = DateTime(now.year, now.month, now.day)
              .difference(DateTime(last.year, last.month, last.day))
              .inDays;
          if (gap == 1) {
            streak += 1;
          } else if (gap > 1) {
            streak = 1;
          }
        } else {
          streak = 1;
        }
      } else {
        streak = 1;
      }
      lastPracticeDate = today;
    }

    xp += earnedXp;
    dailyXp += earnedXp;
    final prev = bestScores[lessonId] ?? 0;
    if (averageScore > prev) bestScores[lessonId] = averageScore;
    if (averageScore >= 55) {
      completedLessons.add(lessonId);
    }
    notifyListeners();
    await _save();
  }

  Future<void> loseHeart() async {
    if (hearts > 0) hearts -= 1;
    notifyListeners();
    await _save();
  }

  Future<void> refillHearts() async {
    hearts = 5;
    notifyListeners();
    await _save();
  }

  Future<void> resetAll() async {
    xp = 0;
    streak = 0;
    hearts = 5;
    dailyXp = 0;
    lastPracticeDate = null;
    completedLessons.clear();
    bestScores.clear();
    notifyListeners();
    await _save();
  }

  int completedCount() => completedLessons.length;

  double overallProgress() {
    final total = allLessons().length;
    if (total == 0) return 0;
    return completedLessons.length / total;
  }
}

class SoftButton extends StatelessWidget {
  final String label;
  final VoidCallback? onPressed;
  final Color color;
  final Color dark;
  final IconData? icon;
  final bool outlined;

  const SoftButton({
    super.key,
    required this.label,
    required this.onPressed,
    this.color = AppColors.green,
    this.dark = AppColors.greenDark,
    this.icon,
    this.outlined = false,
  });

  @override
  Widget build(BuildContext context) {
    final enabled = onPressed != null;
    return GestureDetector(
      onTap: onPressed,
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 120),
        height: 54,
        decoration: BoxDecoration(
          color: outlined
              ? Colors.white
              : (enabled ? color : AppColors.line),
          borderRadius: BorderRadius.circular(16),
          border: Border.all(
            color: outlined
                ? AppColors.line
                : (enabled ? dark : const Color(0xFFCFCFCF)),
            width: 2,
          ),
          boxShadow: enabled && !outlined
              ? [
                  BoxShadow(
                    color: dark,
                    offset: const Offset(0, 4),
                    blurRadius: 0,
                  ),
                ]
              : null,
        ),
        child: Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            if (icon != null) ...[
              Icon(
                icon,
                color: outlined ? AppColors.ink : Colors.white,
              ),

            ],

              label.toUpperCase(),

      expect(memory.vocabulary, hasLength(1));

    });

      await memory.recordVocabulary(word: 'went');
      await memory.recordCorrection(
        original: 'I goed',
        corrected: 'I went',
      );

      expect(memory.vocabulary, hasLength(1));
      expect(memory.corrections, hasLength(1));
      expect(memory.items, hasLength(2));
    });

    test('reviewing an item schedules it for the future', () async {

      await memory.load();
      await memory.recordVocabulary(word: 'carefully');

      final item = memory.vocabulary.single;
